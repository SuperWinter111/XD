	long timestamp = 0;
	float ip_lon;
	float ip_lat;
	float ip_alt;

	float tgt_alt;  //目标高度

	float tgt_lon; 
	float tgt_lat;
	char guide_type; //末端制导方式 0-无引导，1-烟雾，2-激光，3-红外
	unsigned int laser_code;  //激光编码
	float tgt_to_nearest_friend_dis; //目标到最近友方单位距离
	char tgt_to_nearest_friend_yaw; //目标到最近友方单位航向 0-无效，1-北，2-东北，3-东，4-东南，5-南，6-西南，7-西，8-西北
	float ep_lon;
	float ep_lat;
	float ep_alt;

  };
